async function buscarYDescargar() { const folioInput = document.getElementById('folioInput').value.trim().toUpperCase(); const errorMsg = document.getElementById('errorMsg'); const loadingMsg = document.getElementById('loadingMsg'); errorMsg.style.display = 'none'; loadingMsg.style.display = 'none'; if (!folioInput) { errorMsg.textContent = "Por favor ingresa un folio válido."; errorMsg.style.display = 'block'; return; } loadingMsg.style.display = 'block'; loadingMsg.textContent = "Generando tu documento oficial en PDF..."; try { const url = `${WEB_APP_URL}?folio=${encodeURIComponent(folioInput)}`; const response = await fetch(url, { method: "GET", redirect: "follow" }); const result = await response.json(); if (result.status === "success") { // Descargar el archivo PDF directamente generado desde Google Docs const link = document.createElement('a'); link.href = "data:application/pdf;base64," + result.pdfBase64; link.download = result.filename; document.body.appendChild(link); link.click(); document.body.removeChild(link); loadingMsg.style.display = 'none'; } else if (result.status === "not_found") { loadingMsg.style.display = 'none'; errorMsg.textContent = "Folio no encontrado. Verifica tus datos."; errorMsg.style.display = 'block'; } else { loadingMsg.style.display = 'none'; errorMsg.textContent = "Error: " + (result.message || "No se pudo procesar la solicitud."); errorMsg.style.display = 'block'; } } catch (error) { console.error("Error al conectar:", error); loadingMsg.style.display = 'none'; errorMsg.textContent = "Error de conexión con la base de datos."; errorMsg.style.display = 'block'; } }
